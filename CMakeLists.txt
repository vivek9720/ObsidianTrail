cmake_minimum_required(VERSION 3.16)
project(ObsidianTrail VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB_RECURSE OBSIDIANTRAIL_SRC CONFIGURE_DEPENDS src/*.cpp)
add_library(obsidiantrail STATIC ${OBSIDIANTRAIL_SRC})
target_include_directories(obsidiantrail PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(obsidiantrail PRIVATE /W4)
else()
  target_compile_options(obsidiantrail PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(otinspect tools/otinspect.cpp)
target_link_libraries(otinspect PRIVATE obsidiantrail)
add_executable(obsidiantrail_tests tests/test_all.cpp)
target_link_libraries(obsidiantrail_tests PRIVATE obsidiantrail)
enable_testing()
add_test(NAME obsidiantrail_tests COMMAND obsidiantrail_tests)
